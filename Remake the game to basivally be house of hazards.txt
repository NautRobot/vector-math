       // --- CLASSES ---
       class Player {
           constructor(id, color, inputs) {
               this.id = id;
               this.color = color;
               this.inputs = inputs;
               this.w = 30; this.h = 30;
               this.reset();
           }
           reset() {
               this.x = 485; 
               this.y = 580; // Start at bottom middle (Stairs bottom)
               this.vx = 0; this.vy = 0;
               this.grounded = false;
               this.dead = false;
               this.pushCooldown = 0;
               
               this.accel = 2200;
               this.friction = 0.80; // slippery
               this.grav = 1600;
               this.jumpForce = 750;
               this.knockback = false;
           }
           die() {
               if(this.dead) return;
               this.dead = true;
               screenShake = 12;
               for(let i=0; i<30; i++) particles.push(new Particle(this.x + this.w/2, this.y + this.h/2, this.color));
               checkRoundEnd();
           }
           update(dt, other) {
               if (this.dead || gameState !== 'PLAYING') return;

               if (this.pushCooldown > 0) this.pushCooldown -= dt;

               // Shove Mechanic
               if (this.inputs.action && this.pushCooldown <= 0 && !other.dead) {
                   let dx = other.x - this.x;
                   let dy = other.y - this.y;
                   if (Math.hypot(dx, dy) < 70) {
                       let dir = (dx > 0) ? 1 : -1;
                       other.vx = dir * 1800; 
                       other.vy = -400;
                       this.pushCooldown = 1.0;
                       screenShake = 5;
                       for(let i=0; i<8; i++) particles.push(new Particle(this.x+this.w/2, this.y+this.h/2, "#fff"));
                   }
               }

               if (this.inputs.left) this.vx -= this.accel * dt;
               if (this.inputs.right) this.vx += this.accel * dt;
               
               this.vx *= this.friction;
               this.vy += this.grav * dt;

               if (this.inputs.up && this.grounded) {
                   this.vy = -this.jumpForce;
                   this.grounded = false;
               }

               this.x += this.vx * dt;
               this.resolveCollision('x');
               this.y += this.vy * dt;
               this.grounded = false;
               this.resolveCollision('y');

               // Check Traps
               for(let t of traps) if (t.checkHit(this)) this.die();
               for(let p of projectiles) if(this.x < p.x+p.w && this.x+this.w > p.x && this.y < p.y+p.h && this.y+this.h > p.y) this.die();

               // Check Goal
               if (currentTask) {
                   let tgt = currentTask.target;
                   if (this.x < tgt.x+tgt.w && this.x+this.w > tgt.x && this.y < tgt.y+tgt.h && this.y+this.h > tgt.y) {
                       winRound(this.id);
                   }
               }
           }
           resolveCollision(axis) {
               for (let s of structures) {
                   if (this.x < s.x + s.w && this.x + this.w > s.x && this.y < s.y + s.h && this.y + this.h > s.y) {
                       // Jump-through platforms logic
                       if (s.type === 'plat') {
                           if (axis === 'y' && this.vy > 0 && this.y + this.h - this.vy * 0.016 <= s.y + 10) {
                               this.y = s.y - this.h;
                               this.vy = 0;
                               this.grounded = true;
                           }
                       } else {
                           // Solid wall/floor
                           if (axis === 'x') {
                               if (this.vx > 0) this.x = s.x - this.w;
                               else if (this.vx < 0) this.x = s.x + s.w;
                               this.vx = 0;
                           } else {
                               if (this.vy > 0) {
                                   this.y = s.y - this.h;
                                   this.grounded = true;
                               } else if (this.vy < 0) {
                                   this.y = s.y + s.h;
                               }
                               this.vy = 0;
                           }
                       }
                   }
               }
           }
           draw() {
               if(this.dead) return;
               ctx.fillStyle = this.color;
               ctx.fillRect(this.x, this.y, this.w, this.h);
               // Eyes
               ctx.fillStyle = "#fff";
               let faceX = (this.vx > 0) ? 14 : (this.vx < 0) ? 6 : 10;
               ctx.fillRect(this.x + faceX, this.y + 6, 4, 6);
               ctx.fillRect(this.x + faceX + 8, this.y + 6, 4, 6);
               
               // Cooldown ring
               if(this.pushCooldown > 0) {
                   ctx.fillStyle = "rgba(255,255,255,0.7)";
                   ctx.fillRect(this.x, this.y + this.h - 6, this.w * (1 - this.pushCooldown), 6);
               }
           }
       }

       class Particle {
           constructor(x, y, color) {
               this.x = x; this.y = y;
               this.color = color;
               this.size = Math.random() * 8 + 4;
               this.life = 1.0;
               let angle = Math.random() * Math.PI * 2;
               let speed = Math.random() * 400 + 100;
               this.vx = Math.cos(angle) * speed;
               this.vy = Math.sin(angle) * speed;
           }
           update(dt) {
               this.x += this.vx * dt;
               this.y += this.vy * dt;
               this.vy += 1000 * dt; // gravity
               this.life -= dt * 1.5;
           }
           draw() {
               if(this.life <= 0) return;
               ctx.globalAlpha = this.life;
               ctx.fillStyle = this.color;
               ctx.fillRect(this.x, this.y, this.size, this.size);
               ctx.globalAlpha = 1.0;
           }
       }

       // --- TRAPS ---
       class Trap {
           constructor(x, y, botId) { this.x = x; this.y = y; this.botId = botId; this.state = 'idle'; this.timer = 0; }
           trigger() { if (this.state === 'idle') { this.state = 'warn'; this.timer = 0.5; } } // Default warn time
           update(dt) {}
           draw() {}
           checkHit(p) { return false; }
           distToTarget() {
               // Bots check how close players are to decide if they should trigger
               let d = 9999;
               if(!p1.dead) d = Math.min(d, Math.hypot(p1.x - this.x, p1.y - this.y));
               if(!p2.dead) d = Math.min(d, Math.hypot(p2.x - this.x, p2.y - this.y));
               return d;
           }
       }

       class FallingLamp extends Trap {
           constructor(x, y) { super(x, y, 1); this.w = 40; this.h = 40; this.baseY = y; }
           update(dt) {
               if (this.state === 'warn') {
                   this.timer -= dt;
                   if (this.timer <= 0) { this.state = 'fall'; this.vy = 0; }
               } else if (this.state === 'fall') {
                   this.vy += 2000 * dt;
                   this.y += this.vy * dt;
                   if (this.y > 600) { this.state = 'reset'; this.timer = 2.0; screenShake = 8; }
               } else if (this.state === 'reset') {
                   this.timer -= dt;
                   if (this.timer <= 0) { this.y = this.baseY; this.state = 'idle'; }
               }
           }
           checkHit(p) { return this.state === 'fall' && (p.x < this.x+this.w && p.x+p.w > this.x && p.y+p.h > this.y && p.y < this.y+this.h); }
           draw() {
               ctx.fillStyle = (this.state === 'warn' && Math.floor(this.timer*15)%2===0) ? "#FF0000" : "#F4D03F";
               ctx.beginPath();
               ctx.moveTo(this.x + this.w/2, this.y);
               ctx.lineTo(this.x + this.w, this.y + this.h);
               ctx.lineTo(this.x, this.y + this.h);
               ctx.fill();
               // Cord
               ctx.strokeStyle = "#222"; ctx.lineWidth = 2;
               ctx.beginPath(); ctx.moveTo(this.x + this.w/2, 0); ctx.lineTo(this.x + this.w/2, this.y); ctx.stroke();
           }
       }

       class Toaster extends Trap {
           constructor(x, y) { super(x, y, 2); this.w = 50; this.h = 40; }
           update(dt) {
               if (this.state === 'warn') {
                   this.timer -= dt;
                   if (this.timer <= 0) {
                       this.state = 'cooldown'; this.timer = 1.5;
                       projectiles.push({x: this.x+10, y: this.y-10, w: 20, h: 20, vx: -300, vy: -600, color: '#C0392B'});
                       projectiles.push({x: this.x+20, y: this.y-10, w: 20, h: 20, vx: -150, vy: -700, color: '#C0392B'});
                   }
               } else if (this.state === 'cooldown') {
                   this.timer -= dt;
                   if (this.timer <= 0) this.state = 'idle';
               }
           }
           draw() {
               ctx.fillStyle = (this.state === 'warn') ? "#E74C3C" : "#95A5A6";
               ctx.fillRect(this.x, this.y, this.w, this.h);
               ctx.fillStyle = "#2C3E50";
               ctx.fillRect(this.x+10, this.y-5, 30, 10);
           }
       }

       class Roomba extends Trap {
           constructor(x, y) { super(x, y, 3); this.w = 50; this.h = 20; this.vx = 150; }
           trigger() { this.vx *= -1; } // Bots reverse it randomly
           update(dt) {
               this.x += this.vx * dt;
               if (this.x < 550) { this.x = 550; this.vx = Math.abs(this.vx); }
               if (this.x > 900) { this.x = 900; this.vx = -Math.abs(this.vx); }
           }
           checkHit(p) { return (p.x < this.x+this.w && p.x+p.w > this.x && p.y+p.h > this.y && p.y < this.y+this.h); }
           draw() {
               ctx.fillStyle = "#34495E";
               ctx.beginPath();
               ctx.ellipse(this.x + this.w/2, this.y + this.h/2, this.w/2, this.h/2, 0, 0, Math.PI*2);
               ctx.fill();
               ctx.fillStyle = "#E74C3C";
               ctx.fillRect(this.vx > 0 ? this.x + 35 : this.x + 10, this.y + 5, 5, 5); // Eye
           }
       }

       class Cabinet extends Trap {
           constructor(x, y, dir) { super(x, y, 4); this.w = 10; this.h = 80; this.dir = dir; }
           update(dt) {
               if (this.state === 'warn') {
                   this.timer -= dt;
                   if (this.timer <= 0) { this.state = 'swing'; this.timer = 1.0; }
               } else if (this.state === 'swing') {
                   this.timer -= dt;
                   if (this.timer <= 0) this.state = 'idle';
               }
           }
           checkHit(p) { 
               if(this.state !== 'swing') return false;
               let hitBox = { x: this.dir===1 ? this.x : this.x - 70, y: this.y, w: 80, h: this.h };
               return (p.x < hitBox.x+hitBox.w && p.x+p.w > hitBox.x && p.y+p.h > hitBox.y && p.y < hitBox.y+hitBox.h);
           }
           draw() {
               ctx.fillStyle = "#795548";
               ctx.fillRect(this.x, this.y, this.w, this.h);
               if (this.state === 'swing' || (this.state === 'warn' && Math.floor(this.timer*15)%2===0)) {
                   ctx.fillStyle = "rgba(121, 85, 72, 0.8)";
                   ctx.fillRect(this.dir===1 ? this.x : this.x - 70, this.y, 80, this.h); // open door
               }
           }
       }

       class Puddle extends Trap {
           constructor(x, y) { super(x, y, 5); this.w = 100; this.h = 10; }
           trigger() { if(this.state === 'idle') { this.state = 'active'; this.timer = 3.0; } }
           update(dt) {
               if (this.state === 'active') {
                   this.timer -= dt;
                   if (this.timer <= 0) this.state = 'idle';
                   
                   // Apply ice physics
                   [p1, p2].forEach(p => {
                       if (p.x+p.w/2 > this.x && p.x+p.w/2 < this.x+this.w && p.y+p.h >= this.y - 5) {
                           p.friction = 0.99; // Slippery!
                       } else {
                           p.friction = 0.80; // Normal
                       }
                   });
               }
           }
           draw() {
               if (this.state === 'active') {
                   ctx.fillStyle = "rgba(52, 152, 219, 0.7)";
                   ctx.beginPath();
                   ctx.ellipse(this.x + this.w/2, this.y + this.h/2, this.w/2, this.h/2, 0, 0, Math.PI*2);
                   ctx.fill();
               } else {
                   // Dripping faucet
                   ctx.fillStyle = "#aaa";
                   ctx.fillRect(this.x + this.w/2 - 5, this.y - 40, 10, 20);
               }
           }
       }

       let traps = [];

       function initTraps() {
           traps = [
               new FallingLamp(200, 100),             // Living room
               new Toaster(750, 600),                 // Kitchen
               new Roomba(600, 620),                  // Kitchen floor
               new Cabinet(100, 150, 1),              // Bedroom
               new Puddle(750, 290),                  // Bathroom sink
               new FallingLamp(800, 50)               // Bathroom light
           ];
       }

       function runBotLogic(dt) {
           traps.forEach(t => {
               if (t.state === 'idle') {
                   let dist = t.distToTarget();
                   if (dist < 300 && Math.random() < 0.03) {
                       t.trigger();
                   }
               }
           });
       }

       // --- GAME FLOW ---
       function setTask() {
           currentTask = TASKS[Math.floor(Math.random() * TASKS.length)];
           document.getElementById('task-display').innerText = `TASK: ${currentTask.text}`;
       }

       function startRound() {
           setTask();
           p1.reset();
           p2.reset();
           projectiles = [];
           particles = [];
           initTraps();
           
           gameState = 'COUNTDOWN';
           roundTimer = 3.0;
           document.getElementById('overlay-msg').style.display = 'block';
           document.getElementById('msg-title').innerText = "GET READY";
           document.getElementById('msg-title').style.color = "#FFC107";
       }

       function checkRoundEnd() {
           if (p1.dead && p2.dead) {
               endRound("NOBODY WINS", "Both players were eliminated!");
           }
       }

       function winRound(playerId) {
           if (playerId === 1) { scores.p1++; document.getElementById('score-p1').innerText = scores.p1; }
           if (playerId === 2) { scores.p2++; document.getElementById('score-p2').innerText = scores.p2; }
           
           if (scores.p1 >= WIN_SCORE || scores.p2 >= WIN_SCORE) {
               endGame(playerId);
           } else {
               endRound(`PLAYER ${playerId} WINS`, `Task Completed: ${currentTask.text}`);
           }
       }

       function endRound(title, desc) {
           gameState = 'ROUND_OVER';
           roundTimer = 3.0;
           const msg = document.getElementById('overlay-msg');
           msg.style.display = 'block';
           document.getElementById('msg-title').innerText = title;
           document.getElementById('msg-title').style.color = (title.includes('1')) ? 'var(--p1-color)' : (title.includes('2')) ? 'var(--p2-color)' : '#fff';
           document.getElementById('msg-desc').innerText = desc;
       }

       function endGame(winnerId) {
           gameState = 'GAME_OVER';
           const msg = document.getElementById('overlay-msg');
           msg.style.display = 'block';
           document.getElementById('msg-title').innerText = `PLAYER ${winnerId} WINS THE GAME!`;
           document.getElementById('msg-title').style.color = winnerId === 1 ? 'var(--p1-color)' : 'var(--p2-color)';
           document.getElementById('msg-desc').innerText = "Refresh to play again.";
       }

       function gameLoop(time) {
           let dt = (time - lastTime) / 1000;
           lastTime = time;
           if (dt > 0.1) dt = 0.1;

           if (gameState === 'COUNTDOWN') {
               roundTimer -= dt;
               document.getElementById('msg-desc').innerText = Math.ceil(roundTimer);
               if (roundTimer <= 0) {
                   gameState = 'PLAYING';
                   document.getElementById('overlay-msg').style.display = 'none';
               }
           } else if (gameState === 'ROUND_OVER') {
               roundTimer -= dt;
               if (roundTimer <= 0) startRound();
           }

           if (gameState === 'PLAYING') {
               p1.update(dt, p2);
               p2.update(dt, p1);
               runBotLogic(dt);
           }

           traps.forEach(t => t.update(dt));

           for(let i=projectiles.length-1; i>=0; i--) {
               let p = projectiles[i];
               p.x += p.vx * dt;
               p.y += p.vy * dt;
               p.vy += 1200 * dt; // gravity for toast/projectiles
               
               let hitWall = structures.some(s => p.x < s.x+s.w && p.x+p.w > s.x && p.y < s.y+s.h && p.y+p.h > s.y);
               if (hitWall || p.y > 700) projectiles.splice(i,1);
           }

           for(let i=particles.length-1; i>=0; i--) {
               particles[i].update(dt);
               if (particles[i].life <= 0) particles.splice(i,1);
           }

           if (screenShake > 0) screenShake -= 40 * dt;

           // --- DRAWING ---
           ctx.fillStyle = "#111"; // Default bg
           ctx.fillRect(0, 0, cvs.width, cvs.height);

           ctx.save();
           if (screenShake > 0) {
               ctx.translate((Math.random()-0.5)*screenShake, (Math.random()-0.5)*screenShake);
           }

           // Draw Rooms (Backgrounds)
           rooms.forEach(r => {
               ctx.fillStyle = r.color;
               ctx.fillRect(r.x, r.y, r.w, r.h);
               // Room Name
               ctx.fillStyle = "rgba(0,0,0,0.1)";
               ctx.font = "bold 40px sans-serif";
               ctx.fillText(r.name.toUpperCase(), r.x + 20, r.y + 60);
           });

           // Target Highlight
           if (currentTask && gameState === 'PLAYING') {
               let tgt = currentTask.target;
               ctx.fillStyle = `rgba(46, 204, 113, ${0.3 + Math.sin(time/200)*0.1})`; // pulsing green
               ctx.fillRect(tgt.x, tgt.y, tgt.w, tgt.h);
               ctx.strokeStyle = "#2ECC71";
               ctx.lineWidth = 4;
               ctx.strokeRect(tgt.x, tgt.y, tgt.w, tgt.h);
               
               ctx.font = "30px sans-serif";
               ctx.fillText(currentTask.icon, tgt.x + tgt.w/2 - 15, tgt.y - 10 + Math.sin(time/150)*5);
           }

           // Decor / Objects
           // Fridge
           ctx.fillStyle = "#ecf0f1"; ctx.fillRect(880, 510, 80, 130);
           ctx.fillStyle = "#bdc3c7"; ctx.fillRect(890, 510, 2, 130); // split
           // Couch
           ctx.fillStyle = "#8e44ad"; ctx.fillRect(50, 600, 120, 40); ctx.fillRect(50, 570, 20, 70); ctx.fillRect(150, 570, 20, 70);
           // TV
           ctx.fillStyle = "#2c3e50"; ctx.fillRect(250, 550, 80, 50); ctx.fillStyle = "#3498db"; ctx.fillRect(255, 555, 70, 40);
           // Bed
           ctx.fillStyle = "#8b4513"; ctx.fillRect(50, 280, 120, 20); ctx.fillStyle = "#3498db"; ctx.fillRect(50, 260, 90, 20); ctx.fillStyle = "#fff"; ctx.fillRect(140, 260, 30, 20);
           // Toilet/Sink
           ctx.fillStyle = "#fff"; ctx.fillRect(850, 250, 30, 50); ctx.fillRect(850, 230, 15, 20); // toilet
           ctx.fillRect(750, 260, 40, 40); // sink base

           // Structures
           structures.forEach(s => {
               ctx.fillStyle = s.type === 'plat' ? "#555" : "#333";
               ctx.fillRect(s.x, s.y, s.w, s.h);
               if (s.type === 'floor') {
                   ctx.fillStyle = "#4a3f35"; // wood color
                   ctx.fillRect(s.x, s.y, s.w, 10);
               } else if (s.type === 'plat') {
                   ctx.fillStyle = "#777";
                   ctx.fillRect(s.x, s.y, s.w, 4);
               }
           });

           // Entities
           traps.forEach(t => t.draw());
           
           projectiles.forEach(p => {
               ctx.fillStyle = p.color;
               ctx.fillRect(p.x, p.y, p.w, p.h);
           });

           p1.draw();
           p2.draw();
           particles.forEach(p => p.draw());

           ctx.restore();

           requestAnimationFrame(gameLoop);
       }

       decoy.addEventListener('dblclick', () => {
           decoy.style.display = 'none';
           wrapper.style.display = 'block';
           lastTime = performance.now();
           startRound();
           requestAnimationFrame(gameLoop);
       });
   </script>
